function calculateRisk() { // Get user inputs const T_celsius = parseFloat(document.getElementById("temperature").value); const P = parseFloat(document.getElementById("pressure").value); const A = parseFloat(document.getElementById("A").value); const B = parseFloat(document.getElementById("B").value); // Convert Celsius to Kelvin const T = T_celsius + 273.15; // Validate inputs if (isNaN(T) || isNaN(P) || isNaN(A) || isNaN(B)) { alert("Please enter valid numeric values for all fields."); return; } // Calculate risk using final formula: R = P / e^((A-B)/T) - 1 const denominator = Math.exp((A - B) / T); const R = (P / denominator) - 1; // Display results document.getElementById("result").innerHTML = `Risk of Hydrate Formation Score: ${R.toFixed(4)}`; document.getElementById("resultPercentage").innerHTML = `Risk of Hydrate Formation Percentage: ${(R * 100).toFixed(2)}%`; // Simple suggestions let suggestions = ""; if (R > 0) { suggestions = "Suggestions:
    "; suggestions += "
  • Reduce pressure or increase temperature to lower hydrate risk.
    • "; suggestions += "
"; } else { suggestions = "No hydrate formation risk."; } document.getElementById("suggestions").innerHTML = suggestions; // Optionally display formula if you want document.getElementById("formulas").innerHTML = `

Hydrate Formation Risk Calculation:

R = P / e(A - B) / T - 1

`; }
